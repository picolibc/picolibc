[binaries]
c = ['clang', '--target=hexagon-unknown-none-elf', '--cstdlib=picolibc', '-mv68', '-nostdlib', '-fuse-init-array', '-nostdlibinc', '-G0']
cpp = ''
c_ld = 'eld'
ar = 'ar'
as = 'as'
nm = 'nm'
strip = 'strip'
objcopy = 'objcopy'
# only needed to run tests
exe_wrapper = ['env', 'run-hexagon-linux']

[host_machine]
system='linux'
cpu_family='hexagon'
cpu='hexagon'
endian='little'

[properties]
# hardcode runtime library, otherwise meson build configuration will add '-lgcc'
# use -Dc_link_args=-L$(clang -print-resource-dir)/lib/hexagon-unknown-none-elf flag while calling hexagon configure script
# so that clang can find builtins library
librt = '-lclang_rt.builtins'
skip_sanity_check = true
needs_exe_wrapper = true
link_spec = '--build-id=none'
# sdata needs to be 64byte aligned for Hexagon
sdata_alignment = '64'
# Hexagon Linux uses 64 KiB pages; user-mode QEMU rejects 4 KiB-aligned
# PT_LOAD segments with EINVAL while mapping the executable.
default_linux_alignment = '0x10000'
target_c_args = ['-static', '-fno-pic', '-fno-PIE', '-nostdlib', '-nostartfiles', '-Wl,--build-id=none', '-Wl,-z,max-page-size=0x10000']
# default crt0/oslib/linkerscript so the build system picks linux variants
# even when has_semihost is true (hexagon also has a semihost backend)
crt0_default             = 'crt0-linux'
oslib_default            = 'linux'
linkerscript_default_c   = 'picolibc_linux.ld'
linkerscript_default_cpp = 'picolibcpp_linux.ld'
# memory layout: same values as cross-clang-hexagon.txt so flash and ram stay
# within the 256KB GP-relative range (kept for linker script address validity)
default_flash_addr = '0x00100000'
default_flash_size = '0x00400000'
default_ram_addr   = '0x00500000'
default_ram_size   = '0x00800000'
