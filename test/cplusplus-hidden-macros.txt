# Macros which picolibc headers define in C but not in C++.
#
# C++ code may use any name which C++ doesn't define as a macro, for
# example as the name of a member function, so these C macros aren't
# defined in C++. math.h provides overloaded inline functions with the
# same names instead, except for isinff, isnanf, isinfl and isnanl,
# which are library functions in both C and C++.
#
# math.h also doesn't declare the legacy int isinf(double) and int
# isnan(double) functions from C++11 on, as they conflict with the
# bool overloads in <cmath>.
#
# test/meson.build runs scripts/check-cplusplus-macros to check that
# none of these is defined in any C++ version the compiler supports.
#
# header   macro

# C23 classification macros
math.h     iscanonical
math.h     iseqsig
math.h     issignaling
math.h     issubnormal
math.h     iszero

# picolibc suffixed classification macros
math.h     isfinitef
math.h     isfinitel
math.h     isinff
math.h     isinfl
math.h     isnanf
math.h     isnanl
math.h     isnormalf
math.h     isnormall
math.h     iszerof
math.h     iszerol
